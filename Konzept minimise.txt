def backtracking_line_search(x, p, E_old, energy_func, alpha0=1e-3):
    alpha = alpha0

    for _ in range(50):
        x_new = x + alpha * p
        E_new = energy_func(x_new)

        if E_new < E_old:
            return alpha

        alpha *= 0.5

    return alpha
